\subsubsection*{Phần I. Trắc nghiệm}
\begin{baitap}
Nghiệm của phương trình $2x+4=0$ là:
\begin{enumerate}[A.]
\item $x=2$.
\item $x=-2$.
\item $x=0$.
\item $x=4$.
\end{enumerate}
\begin{traloi}
\dapan{B}
Ta có $2x=-4$, suy ra $x=-2$.
\end{traloi}
\end{baitap}
\subsubsection*{Phần II. Đúng hoặc sai}
\begin{baitap}
Xét hàm số $y=x^2$.
\begin{enumerate}[a)]
\item Đồ thị đi qua gốc tọa độ.
\item Giá trị tại $x=2$ bằng $2$.
\item Hàm số có giá trị nhỏ nhất bằng $0$.
\item Hàm số nhận giá trị âm tại $x=-1$.
\end{enumerate}
\begin{traloi}
\dapan{Đ,S,Đ,S}
Ta có $0^2=0$, $2^2=4$, $x^2\ge0$ và $(-1)^2=1$.
\end{traloi}
\end{baitap}
\subsubsection*{Phần III. Trả lời ngắn}
\begin{baitap}
Tính $5:2$.
\begin{traloi}
\dapan{2,5}
Chia $5$ cho $2$ được $2{,}5$.
\end{traloi}
\end{baitap}
